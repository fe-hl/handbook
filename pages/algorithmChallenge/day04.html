24.两两交换链表中的节点 ​

js
const swapPairs = function (head) {
  let dummy = {
    next: head,
  };
  let cur = dummy;
  while (cur.next && cur.next.next) {
    let temp1 = cur.next;
    let temp2 = cur.next.next.next;
    // 交换
    cur.next = cur.next.next;
    cur.next.next = temp1;
    temp1.next = temp2;
    // 移动指针到需要交互节点的前一个位置
    cur = temp1;
  }
  return dummy.next;
};

19.删除链表的倒数第 N 个结点 ​

js
const removeNthFromEnd = function (head, n) {
  let dummy = {
    next: head,
  };
  let slow = dummy;
  let fast = dummy;
  // 先让fast指针走n步
  for (let i = 0; i <= n; i++) {
    fast = fast.next;
  }
  // slow和fast同时走，直到fast走到链表末尾
  // 此时slow的下一个节点就是需要删除的节点
  while (fast) {
    fast = fast.next;
    slow = slow.next;
  }
  slow.next = slow.next.next;
  return dummy.next;
};

160.相交链表 ​

js
// 双指针法，让两个指针分别走完两个链表，如果两个指针相遇，则说明两个链表相交
// 如果两个指针没有相遇，则说明两个链表不相交,最会在null处相遇
const getIntersectionNode = function (headA, headB) {
  if (headA === null || headB === null) return null;
  let pA = headA;
  let pB = headB;
  while (pA != pB) {
    if (pA) {
      pA = pA.next;
    } else {
      pA = headB;
    }
    if (pB) {
      pB = pB.next;
    } else {
      pB = headA;
    }
  }
  return pA;
};

142.环形链表 2 ​

js
const detectCycle = function (head) {
  let slow = head;
  let fast = head;
  while (fast && fast.next) {
    fast = fast.next.next;
    slow = slow.next;
    if (fast === slow) {
      // 有环
      slow = head;
      while (slow !== fast) {
        slow = slow.next;
        fast = fast.next;
      }
      // 返回环的入口
      return slow;
    }
  }
  return null;
};
