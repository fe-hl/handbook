209.长度最小的子数组 ​

这题适合直接用滑动窗口。右边界负责把窗口撑开，左边界负责在满足条件时尽量收缩，这样就能在每次满足 sum >= target 时，顺手更新一次最短长度。

js
const minSubArrayLen = function (target, nums) {
  let slow = 0;
  let result = Infinity;
  let sum = 0;
  for (let fast = 0; fast < nums.length; fast++) {
    sum += nums[fast];
    while (sum >= target) {
      result = Math.min(result, fast + 1 - slow);
      sum -= nums[slow];
      slow++;
    }
  }
  return result === Infinity ? 0 : result;
};

59.螺旋矩阵 II ​

这题本质上是在模拟“按圈填数”。每一轮依次处理上、右、下、左四条边，处理完后同步收缩四个边界，直到整个矩阵被填满。

js
const generateMatrix = function (n) {
  let top = 0;
  let left = 0;
  let bottom = n - 1;
  let right = n - 1;
  let count = 1;
  const result = Array.from({ length: n }, () => new Array(n).fill(0));
  while (count <= n * n) {
    // 填充top,从左到右
    for (let i = left; i <= right; i++) {
      result[top][i] = count++;
    }
    top++;

    // 填充right,从上到下
    for (let i = top; i <= bottom; i++) {
      result[i][right] = count++;
    }
    right--;

    // 填充bottom,从右到左
    for (let i = right; i >= left && top <= bottom; i--) {
      result[bottom][i] = count++;
    }
    bottom--;

    // 填充left,从下到上
    for (let i = bottom; i >= top && left <= right; i--) {
      result[i][left] = count++;
    }
    left++;
  }
  return result;
};
